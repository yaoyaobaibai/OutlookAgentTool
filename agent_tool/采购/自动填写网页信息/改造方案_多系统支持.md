#### 步骤 2: 创建 multi_system_launcher.py（启动器）

这个文件作为统一入口，提供系统选择界面，然后调用对应的程序。

```python
# multi_system_launcher.py
import tkinter as tk
from tkinter import ttk, messagebox
import json
import os
import subprocess
import sys

class SystemLauncher:
    def __init__(self):
        self.root = tk.Tk()
        self.root.title("多系统表单工具 - 启动器")
        self.root.geometry("600x400")
        
        self.base_dir = os.path.dirname(os.path.abspath(__file__))
        self.systems = []
        self.load_systems()
        
        self._create_widgets()
    
    def load_systems(self):
        config_path = os.path.join(self.base_dir, 'system_config.json')
        if os.path.exists(config_path):
            with open(config_path, 'r', encoding='utf-8') as f:
                config = json.load(f)
                self.systems = config.get('systems', [])
    
    def _create_widgets(self):
        # 系统选择
        frame = ttk.Frame(self.root, padding=20)
        frame.pack(fill=tk.BOTH, expand=True)
        
        ttk.Label(frame, text="选择系统:", font=('Arial', 12, 'bold')).pack(pady=10)
        
        self.system_var = tk.StringVar()
        system_names = [f"{s.get('icon', '')} {s.get('system_name', '')}" for s in self.systems]
        combo = ttk.Combobox(frame, textvariable=self.system_var, values=system_names, state='readonly', width=30)
        combo.pack(pady=10)
        combo.current(0)
        
        # 流程选择
        ttk.Label(frame, text="选择流程:", font=('Arial', 12, 'bold')).pack(pady=10)
        
        self.flow_var = tk.StringVar()
        self.flows_by_system = {}
        for system in self.systems:
            system_id = system.get('system_id')
            flows = [f for f in self._load_all_flows() if f.get('system_id') == system_id]
            self.flows_by_system[system_id] = flows
        
        def on_system_change(event):
            idx = combo.current()
            system_id = self.systems[idx]['system_id']
            flows = self.flows_by_system.get(system_id, [])
            flow_names = [f.get('flow_name', '') for f in flows]
            flow_combo['values'] = flow_names
            if flow_names:
                flow_combo.current(0)
        
        combo.bind('<<ComboboxSelected>>', on_system_change)
        
        flow_combo = ttk.Combobox(frame, textvariable=self.flow_var, state='readonly', width=30)
        flow_combo.pack(pady=10)
        
        # 启动按钮
        btn_frame = ttk.Frame(frame)
        btn_frame.pack(pady=20)
        
        ttk.Button(btn_frame, text="启动", command=self._launch).pack(side=tk.LEFT, padx=10)
        ttk.Button(btn_frame, text="退出", command=self.root.quit).pack(side=tk.LEFT, padx=10)
    
    def _load_all_flows(self):
        flows_dir = os.path.join(self.base_dir, 'flows')
        flows = []
        if os.path.exists(flows_dir):
            for file in os.listdir(flows_dir):
                if file.endswith('.json'):
                    with open(os.path.join(flows_dir, file), 'r', encoding='utf-8') as f:
                        flows.append(json.load(f))
        return flows
    
    def _launch(self):
        system_idx = self.system_var.get()
        flow_idx = self.flow_var.get()
        
        # 根据选择启动对应的程序
        # 如果是 CSMS 提案流程，启动 auto_create_proposal.py
        # 否则启动 form_filler.py 并传递配置
        
        subprocess.Popen([sys.executable, "auto_create_proposal.py"], cwd=self.base_dir)
        # 或者
        # subprocess.Popen([sys.executable, "form_filler.py"], cwd=self.base_dir)

if __name__ == '__main__':
    app = SystemLauncher()
    app.root.mainloop()
```

### 方案 2：深度整合（推荐）

**将多系统支持直接集成到 form_filler.py 中**

#### 修改点：

1. **在 form_filler.py 开头添加系统配置加载**
2. **在 GUI 顶部添加系统选择下拉框**
3. **根据系统配置动态加载选择器**

#### 具体修改：

**修改 form_filler.py 的 ConfigManager 类**:

```python
class ConfigManager:
    def __init__(self, config_file="form_config.json", system_id=None):
        self.config_file = config_file
        self.system_id = system_id  # 新增
        self.fields = []
        self.system_selectors = {}  # 新增：系统选择器
        self.load_config()
        if system_id:
            self.load_system_selectors(system_id)  # 新增
    
    def load_system_selectors(self, system_id):
        """加载系统特定选择器"""
        systems_dir = os.path.join(os.path.dirname(__file__), 'systems')
        system_file = os.path.join(systems_dir, f"{system_id}.json")
        
        if os.path.exists(system_file):
            with open(system_file, 'r', encoding='utf-8') as f:
                system_config = json.load(f)
                self.system_selectors = system_config.get('selectors', {})
```

**修改 GUI 添加系统选择**:

在 `_create_widgets()` 方法中添加：

```python
# 系统选择区域（新增）
system_frame = ttk.LabelFrame(self.root, text="系统选择", padding=10)
system_frame.pack(fill=tk.X, padx=10, pady=5)

self.system_var = tk.StringVar()
system_combo = ttk.Combobox(
    system_frame,
    textvariable=self.system_var,
    values=["CSMS", "Acubuy", "SAP"],
    state='readonly',
    width=20
)
system_combo.pack(side=tk.LEFT, padx=10)
system_combo.bind('<<ComboboxSelected>>', self._on_system_changed)

def _on_system_changed(self, event):
    system_name = self.system_var.get()
    system_id = system_name.lower()
    
    # 重新加载配置
    self.config_manager = ConfigManager(system_id=system_id)
    
    # 更新字段列表
    self._refresh_field_list()
```
